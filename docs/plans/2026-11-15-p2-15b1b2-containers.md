# P2-15b1b2 — Restore Person Containers Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Decode the type-2 saved containers (c0/c1/c2) and tail, apply c0→main and
c1→trade for matched persons in `BT_ENTS` order, and prove the O4 oracle.

**Architecture:** A shared `assets::parse_item_record` decodes the proven
`0xC0 + u32 sub + 0x14*sub` record and is reused by the hero parser. The
`game` type-2 decoder consumes the three containers plus tail and requires an
exact body consume. `restore_save_persons` takes an optional `InventoryRuntime*`
and overwrites main/trade with the serialized records via new replace APIs;
duplicate person names are last-wins with a warning; c2 is decoded but not
applied. The `+0x26c0` gate and loot ordering stay in `P2-15b1b3`.

**Tech Stack:** C++20, CMake + Ninja, doctest, SDL2 project layers
(`assets` → `game` → `tools/glgame`).

**Spec:** `docs/specs/2026-11-15-p2-15b1b2-containers-design.md`.

```cpp
#include "gl/assets/save_item.hpp"

#include <new>
#include <string>
#include <utility>

namespace gl::assets {
namespace {
constexpr std::size_t kFixedSize = 0xC4;
constexpr std::size_t kSubrecordSize = 0x14;
constexpr std::size_t kTypeOffset = 0x08;
constexpr std::size_t kFlagsOffset = 0x0C;
constexpr std::size_t kStyleOffset = 0x20;
constexpr std::size_t kModCountOffset = 0xC0;
constexpr std::size_t kItemIdOffset = 0xA8;
constexpr std::size_t kCountOffset = 0xB0;
constexpr std::size_t kDurabilityOffset = 0xB4;

std::uint32_t read_u32(std::span<const std::uint8_t> data, std::size_t offset) {
    return static_cast<std::uint32_t>(data[offset]) |
           (static_cast<std::uint32_t>(data[offset + 1]) << 8) |
           (static_cast<std::uint32_t>(data[offset + 2]) << 16) |
           (static_cast<std::uint32_t>(data[offset + 3]) << 24);
}

Error format_error(std::string message) { return Error{ErrorCode::format, std::move(message)}; }
} // namespace

Result<ItemRecordRead> parse_item_record(std::span<const std::uint8_t> data) {
    if (data.size() < kFixedSize)
        return format_error("item record is shorter than 0xC4");
    const std::uint32_t subcount = read_u32(data, kModCountOffset);
    const std::uint64_t size64 =
        static_cast<std::uint64_t>(kFixedSize) + static_cast<std::uint64_t>(subcount) * kSubrecordSize;
    if (size64 > data.size())
        return format_error("item record subrecords are truncated");
    try {
        ItemRecordRead out;
        out.size = static_cast<std::size_t>(size64);
        out.record.item_id = read_u32(data, kItemIdOffset);
        out.record.count = read_u32(data, kCountOffset);
        out.record.durability = read_u32(data, kDurabilityOffset);
        out.record.type = read_u32(data, kTypeOffset);
        out.record.flags = read_u32(data, kFlagsOffset);
        out.record.style = read_u32(data, kStyleOffset);
        out.record.mod_count = subcount;
        out.record.modifiers.resize(subcount);
        for (std::uint32_t index = 0; index < subcount; ++index) {
            const std::size_t base =
                kFixedSize + static_cast<std::size_t>(index) * kSubrecordSize;
            for (std::size_t word = 0; word < 5; ++word)
                out.record.modifiers[index].words[word] = read_u32(data, base + word * 4);
        }
        return out;
    } catch (const std::bad_alloc &) {
        return format_error("item record allocation failed");
    }
}
} // namespace gl::assets
```

```cpp
void put_u32_at(std::vector<std::uint8_t> &out, std::size_t offset, std::uint32_t value) {
    for (unsigned shift = 0; shift < 32; shift += 8)
        out[offset + shift / 8] = static_cast<std::uint8_t>((value >> shift) & 0xFF);
}

struct ItemSpec {
    std::uint32_t id = 0;
    std::uint32_t count = 1;
    std::uint32_t flags = 0;
    std::uint32_t type = 0;
    std::uint32_t style = 0;
    std::uint32_t durability = 100;
    std::uint32_t mods = 0;
};

void put_item_record(std::vector<std::uint8_t> &out, const ItemSpec &spec) {
    const std::size_t start = out.size();
    out.resize(start + 0xC4, 0);
    put_u32_at(out, start + 0x00, 0x14092248u);
    put_u32_at(out, start + 0x08, spec.type);
    put_u32_at(out, start + 0x0C, spec.flags);
    put_u32_at(out, start + 0x20, spec.style);
    put_u32_at(out, start + 0xA8, spec.id);
    put_u32_at(out, start + 0xB0, spec.count);
    put_u32_at(out, start + 0xB4, spec.durability);
    put_u32_at(out, start + 0xC0, spec.mods);
    for (std::uint32_t index = 0; index < spec.mods; ++index)
        for (std::uint32_t word = 0; word < 5; ++word)
            put_u32(out, index * 10 + word);
}

void put_container(std::vector<std::uint8_t> &out,
                   const std::optional<std::vector<ItemSpec>> &items) {
    if (!items.has_value()) {
        put_u32(out, 0);
        return;
    }
    put_u32(out, 1);
    put_u32(out, static_cast<std::uint32_t>(items->size()));
    for (const ItemSpec &item : *items)
        put_item_record(out, item);
}

std::vector<std::uint8_t>
person_body_with_containers(const PersonValues &values,
                            const std::optional<std::vector<ItemSpec>> &c0,
                            const std::optional<std::vector<ItemSpec>> &c1,
                            const std::optional<std::vector<ItemSpec>> &c2) {
    std::vector<std::uint8_t> out = person_prefix(values);
    put_container(out, c0);
    put_container(out, c1);
    put_container(out, c2);
    out.insert(out.end(), 0x80, 0);
    put_u32(out, 0);
    return out;
}
```

```cpp
std::string container_text(const gl::game::SaveContainer &container) {
    std::string text = container.present ? "1:" : "0:";

TEST_CASE("real data: save container corpus aggregate" * doctest::skip(!game_dir_available())) {
    auto vfs = gl::assets::GameVfs::mount(std::getenv("GL_GAME_DIR"));
    REQUIRE(vfs.has_value());

    auto sdb_bytes = vfs->read("sdb/items/tech_names.sdb");
    REQUIRE(sdb_bytes.has_value());
    auto sdb = gl::assets::Sdb::parse(sdb_bytes.value());
    REQUIRE(sdb.has_value());

    std::size_t type2_records = 0;
    std::size_t item_records = 0;
    std::size_t containers_present = 0;
    std::size_t containers_absent = 0;
    std::size_t zero_count_containers = 0;
    std::size_t c2_items = 0;
    std::size_t selected_records = 0;
    std::set<std::uint32_t> item_ids;

    for (int slot = 0; slot < 28; ++slot) {
        auto payload = gl::assets::parse_save_payload(read_save(slot));
        REQUIRE(payload.has_value());
        CAPTURE(slot);
        REQUIRE(payload->level_tail_index.has_value());
        for (std::size_t index = 0; index < payload->tail.size(); ++index) {
            for (const gl::assets::SaveEntity &entity : payload->tail[index].entities) {
                if (entity.type != 2)
                    continue;
                const auto decoded = gl::game::decode_save_person(entity);
                REQUIRE(decoded.has_value());
                ++type2_records;
                if (index == *payload->level_tail_index)
                    ++selected_records;
                for (const gl::game::SaveContainer &container : decoded->containers) {
                    if (!container.present) {
                        ++containers_absent;
                        continue;
                    }
                    ++containers_present;
                    if (container.items.empty())
                        ++zero_count_containers;
                    item_records += container.items.size();
                    for (const gl::assets::SaveItemRecord &item : container.items)
                        item_ids.insert(item.item_id);
                }
                c2_items += decoded->containers[2].items.size();
            }
        }
    }

    CHECK(type2_records == 14507);
    CHECK(selected_records == 499);
    CHECK(item_records == 115405);
    CHECK(containers_present == 42192);
    CHECK(containers_absent == 1329);
    CHECK(zero_count_containers == 5150);
    CHECK(c2_items == 14156);

    std::size_t unresolved_names = 0;
    std::size_t missing_archives = 0;
    for (std::uint32_t id : item_ids) {
        if (sdb->find(id) == nullptr)
            ++unresolved_names;
        if (!vfs->exists("items/data/" + std::to_string(id) + ".itm"))
            ++missing_archives;
    }
    CHECK(unresolved_names == 0);
    CHECK(missing_archives == 0);
}
```
